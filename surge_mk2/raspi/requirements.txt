# SURGE Mark.2 — Raspberry Pi 5 側 Python 依存
#
# venv は --system-site-packages で作る（gpiozero/lgpio は OS プリインストールを使う）:
#   python3 -m venv --system-site-packages .venv
#   .venv/bin/pip install -r raspi/requirements.txt
#
# gpiozero / lgpio は Raspberry Pi OS に同梱済みのため、ここには書かない。
# （pip 版を入れると OS 版と競合しうる）

# ── Phase 0（通信基盤） ──
pyserial==3.5          # UART 送受信

# ── Phase 0 後半（内部バス・WebSocket） ──
pyzmq                  # 内部バス（PUB/SUB）
msgspec                # メッセージ定義とシリアライズ（内部 msgpack / WS も同じ形）
websockets             # telemetry_node の WS サーバ。純 Python・依存なし
numpy                  # 点群・占有格子・共有メモリ画像の view
mcap                   # ログ記録（logger_node / tools/sfl2mcap.py）。zstandard が付いてくる

# JPEG エンコードは simplejpeg（picamera2 が同梱）を使い、無ければ Pillow に落ちる。
# **Pi では picamera2 経由で入るので明示しない。** Mac でカメラ配信を試すときだけ:
#   pip install pillow

# **`mcap` は io_node には要らない。** 実時間で回るコード（io_node / framelog /
# shm_ring）は stdlib のみで動く方針を維持している。MCAP を書くのは別プロセスの
# logger_node とオフライン変換だけなので、入っていなくても車は走る。
